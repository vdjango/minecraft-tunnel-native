cmake_minimum_required(VERSION 3.24)
project(buggcraft_native VERSION 0.3.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUGGCRAFT_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
option(BUGGCRAFT_LTO "Enable link-time optimization" ON)
option(BUGGCRAFT_BRIDGE "Build native launcher bridge" ON)
include(CTest)
find_package(Threads REQUIRED)
if(APPLE)
  # Bundle crypto into our libraries; never retain a developer's Homebrew path.
  set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
find_package(OpenSSL 3.0 REQUIRED)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/libssh-0.12.2/CMakeLists.txt")
  set(WITH_SERVER ON CACHE BOOL "" FORCE)
  set(WITH_SFTP OFF CACHE BOOL "" FORCE)
  set(WITH_GSSAPI OFF CACHE BOOL "" FORCE)
  set(WITH_ZLIB OFF CACHE BOOL "" FORCE)
  set(WITH_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WITH_FIDO2 OFF CACHE BOOL "" FORCE)
  set(UNIT_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  add_subdirectory(.deps/libssh-0.12.2 libssh-build EXCLUDE_FROM_ALL)
  set(NATIVE_SSH_TARGET ssh)
else()
  find_package(libssh 0.12.2 CONFIG REQUIRED)
  set(NATIVE_SSH_TARGET ssh)
endif()
get_target_property(native_ssh_imported ${NATIVE_SSH_TARGET} IMPORTED)
if(NOT native_ssh_imported)
  if(APPLE)
    set_target_properties(${NATIVE_SSH_TARGET} PROPERTIES INSTALL_RPATH "@loader_path" INSTALL_NAME_DIR "@rpath")
  elseif(UNIX)
    set_target_properties(${NATIVE_SSH_TARGET} PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
endif()

add_executable(buggcraft-core core/main.cpp core/engine.cpp core/worker.cpp
  core/protocol.cpp core/relay.cpp core/stun.cpp core/ipc.cpp)
target_include_directories(buggcraft-core PRIVATE .deps/json/single_include)
target_compile_definitions(buggcraft-core PRIVATE BUGGCRAFT_VERSION="${PROJECT_VERSION}")
target_link_libraries(buggcraft-core PRIVATE ${NATIVE_SSH_TARGET} OpenSSL::Crypto Threads::Threads)
add_custom_command(TARGET buggcraft-core POST_BUILD COMMAND ${CMAKE_COMMAND} -E
  copy_if_different $<TARGET_FILE:${NATIVE_SSH_TARGET}> $<TARGET_FILE_DIR:buggcraft-core>)
if(WIN32)
  target_link_libraries(buggcraft-core PRIVATE ws2_32)
  target_compile_definitions(buggcraft-core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
endif()
if(MSVC)
  target_compile_options(buggcraft-core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(buggcraft-core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(BUGGCRAFT_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT has_ipo OUTPUT ipo_error)
  if(has_ipo)
    set_property(TARGET buggcraft-core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  endif()
endif()
if(BUGGCRAFT_SANITIZERS AND NOT MSVC)
  target_compile_options(buggcraft-core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(buggcraft-core PRIVATE -fsanitize=address,undefined)
endif()
if(APPLE)
  set_target_properties(buggcraft-core PROPERTIES INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
  set_target_properties(buggcraft-core PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS buggcraft-core RUNTIME DESTINATION bin)
install(TARGETS ${NATIVE_SSH_TARGET} RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
if(WIN32)
  get_filename_component(NATIVE_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  set(NATIVE_RUNTIME_DIRS "${NATIVE_COMPILER_DIR};${OPENSSL_ROOT_DIR}/bin;${OPENSSL_INCLUDE_DIR}/../bin")
  # FindOpenSSL uses UNKNOWN imported targets on Windows, so their runtime
  # DLLs are absent from TARGET_RUNTIME_DLLS. Stage the selected toolchain's
  # crypto beside each consumer before dependency scanning can pick a DLL
  # from the build machine's PATH or Windows directories.
  file(GLOB native_crypto_runtime CONFIGURE_DEPENDS
    "${OPENSSL_INCLUDE_DIR}/../bin/libcrypto*.dll"
    "${OPENSSL_INCLUDE_DIR}/../bin/libssl*.dll")
  file(GLOB native_crypto_runtime_debug CONFIGURE_DEPENDS
    "${OPENSSL_INCLUDE_DIR}/../debug/bin/libcrypto*.dll"
    "${OPENSSL_INCLUDE_DIR}/../debug/bin/libssl*.dll")
  include(cmake/StageWindowsCrypto.cmake)
  buggcraft_stage_windows_crypto(buggcraft-core)
  if(MSVC)
    # These are redistributable VC runtime files, not arbitrary System32 DLLs.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
    include(InstallRequiredSystemLibraries)
    install(PROGRAMS ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS} DESTINATION bin)
  endif()
  configure_file(cmake/install-runtime.cmake.in install-runtime.in.cmake @ONLY)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/install-runtime-$<CONFIG>.cmake"
       INPUT "${CMAKE_CURRENT_BINARY_DIR}/install-runtime.in.cmake")
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/install-runtime-$<CONFIG>.cmake")
  buggcraft_install_windows_crypto()
endif()
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/buggcraft-native)
find_file(NATIVE_OPENSSL_LICENSE NAMES copyright LICENSE.txt LICENSE LICENSE.md
  PATHS "${OPENSSL_INCLUDE_DIR}/../share/openssl" "${OPENSSL_INCLUDE_DIR}/.."
  NO_DEFAULT_PATH)
if(NATIVE_OPENSSL_LICENSE)
  install(FILES "${NATIVE_OPENSSL_LICENSE}" DESTINATION share/buggcraft-native/licenses RENAME openssl.txt)
endif()
if(BUGGCRAFT_BRIDGE)
  set(NO_MEDIA ON CACHE BOOL "" FORCE)
  set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
  set(NO_TESTS ON CACHE BOOL "" FORCE)
  set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  add_subdirectory(.deps/libdatachannel datachannel-build EXCLUDE_FROM_ALL)
  if(APPLE)
    set_target_properties(datachannel PROPERTIES INSTALL_RPATH "@loader_path" INSTALL_NAME_DIR "@rpath")
  elseif(UNIX)
    set_target_properties(datachannel PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  add_subdirectory(.deps/zstd/build/cmake zstd-build EXCLUDE_FROM_ALL)
  set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(.deps/zlib-1.3.2 zlib-build EXCLUDE_FROM_ALL)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
  set(protocol_generator "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_protocol_data.py")
  set(protocol_chat "${CMAKE_CURRENT_SOURCE_DIR}/bridge/protocol_data/chat_protocols.json")
  set(protocol_legacy "${CMAKE_CURRENT_SOURCE_DIR}/bridge/protocol_data/legacy_packets.json")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${protocol_generator}" "${protocol_chat}" "${protocol_legacy}")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${protocol_generator}"
    --chat "${protocol_chat}" --legacy "${protocol_legacy}"
    --output "${CMAKE_CURRENT_BINARY_DIR}/generated/protocol_data.hpp"
    COMMAND_ERROR_IS_FATAL ANY)
  add_library(buggcraft-bridge-engine STATIC bridge/runtime.cpp bridge/rpc.cpp bridge/local_game.cpp bridge/game_chat.cpp bridge/ssh.cpp bridge/session.cpp bridge/session_chat.cpp bridge/session_presence.cpp
    bridge/stream.cpp bridge/peer.cpp bridge/peer_health.cpp bridge/codec.cpp bridge/journal.cpp bridge/telemetry.cpp bridge/routing.cpp bridge/route_selector.cpp bridge/modpack.cpp bridge/social.cpp)
  target_include_directories(buggcraft-bridge-engine PRIVATE .deps/json/single_include .deps/zstd/lib "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(buggcraft-bridge-engine PRIVATE BUGGCRAFT_VERSION="${PROJECT_VERSION}")
  target_link_libraries(buggcraft-bridge-engine PUBLIC ${NATIVE_SSH_TARGET} datachannel libzstd_static zlibstatic OpenSSL::Crypto Threads::Threads)
  set_target_properties(buggcraft-bridge-engine PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  add_executable(buggcraft-bridge bridge/main.cpp)
  target_link_libraries(buggcraft-bridge PRIVATE buggcraft-bridge-engine)
  target_compile_definitions(buggcraft-bridge PRIVATE BUGGCRAFT_VERSION="${PROJECT_VERSION}")
  add_library(buggcraft-bridge-library SHARED bridge/c_api.cpp)
  set_target_properties(buggcraft-bridge-library PROPERTIES OUTPUT_NAME buggcraft_bridge CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(WIN32)
    set_target_properties(buggcraft-bridge-library PROPERTIES PREFIX "")
  endif()
  target_include_directories(buggcraft-bridge-library PUBLIC include)
  target_compile_definitions(buggcraft-bridge-library PRIVATE BUGGCRAFT_BRIDGE_EXPORTS)
  target_link_libraries(buggcraft-bridge-library PRIVATE buggcraft-bridge-engine)
  if(WIN32)
    buggcraft_stage_windows_crypto(buggcraft-bridge)
    buggcraft_stage_windows_crypto(buggcraft-bridge-library)
  endif()
  add_custom_command(TARGET buggcraft-bridge POST_BUILD COMMAND ${CMAKE_COMMAND} -E
    copy_if_different $<TARGET_FILE:datachannel> $<TARGET_FILE_DIR:buggcraft-bridge>)
  if(WIN32)
    target_link_libraries(buggcraft-bridge-engine PUBLIC ws2_32 iphlpapi)
    target_compile_definitions(buggcraft-bridge-engine PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
  endif()
  if(MSVC)
    target_compile_options(buggcraft-bridge-engine PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(buggcraft-bridge-engine PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(BUGGCRAFT_LTO AND has_ipo)
    set_property(TARGET buggcraft-bridge-engine buggcraft-bridge buggcraft-bridge-library PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  endif()
  if(APPLE)
    set_target_properties(buggcraft-bridge PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    set_target_properties(buggcraft-bridge-library PROPERTIES INSTALL_RPATH "@loader_path")
  elseif(UNIX)
    set_target_properties(buggcraft-bridge PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    set_target_properties(buggcraft-bridge-library PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  install(TARGETS buggcraft-bridge RUNTIME DESTINATION bin)
  install(TARGETS buggcraft-bridge-library RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
  install(FILES include/buggcraft/bridge.h DESTINATION include/buggcraft)
  install(FILES bindings/python/buggcraft_bridge.py DESTINATION share/buggcraft-native/python)
  install(TARGETS datachannel RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
  if(BUILD_TESTING)
    add_executable(buggcraft-native-tests tests/cpp/main.cpp tests/cpp/journal.cpp tests/cpp/heartbeat.cpp tests/cpp/game_wire.cpp bridge/codec.cpp core/ipc.cpp bridge/journal.cpp)
    target_include_directories(buggcraft-native-tests PRIVATE .deps/json/single_include .deps/zstd/lib)
    target_link_libraries(buggcraft-native-tests PRIVATE ${NATIVE_SSH_TARGET} libzstd_static OpenSSL::Crypto Threads::Threads)
    if(WIN32)
      target_link_libraries(buggcraft-native-tests PRIVATE ws2_32)
      target_compile_definitions(buggcraft-native-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
    endif()
    if(BUGGCRAFT_SANITIZERS AND NOT MSVC)
      foreach(target buggcraft-bridge-engine buggcraft-bridge buggcraft-bridge-library buggcraft-native-tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
      endforeach()
    endif()
    add_test(NAME native_units COMMAND buggcraft-native-tests)
    add_executable(buggcraft-bridge-api-tests tests/cpp/bridge_api.c)
    target_link_libraries(buggcraft-bridge-api-tests PRIVATE buggcraft-bridge-library)
    add_test(NAME bridge_c_abi COMMAND buggcraft-bridge-api-tests)
    set_tests_properties(bridge_c_abi PROPERTIES TIMEOUT 30)
    set_tests_properties(native_units PROPERTIES TIMEOUT 30)
    add_executable(buggcraft-protocol-data-tests tests/cpp/protocol_data.cpp)
    target_include_directories(buggcraft-protocol-data-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(MSVC)
      target_compile_options(buggcraft-protocol-data-tests PRIVATE /utf-8)
    endif()
    add_test(NAME embedded_protocol_tables COMMAND buggcraft-protocol-data-tests "${protocol_chat}" "${protocol_legacy}")
    if(WIN32)
      set_property(TEST native_units APPEND PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:${NATIVE_SSH_TARGET}>")
      set_property(TEST bridge_c_abi APPEND PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:${NATIVE_SSH_TARGET}>")
      foreach(directory IN LISTS NATIVE_RUNTIME_DIRS)
        set_property(TEST native_units APPEND PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${directory}")
        set_property(TEST bridge_c_abi APPEND PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${directory}")
      endforeach()
    endif()
  endif()
endif()
install(DIRECTORY licenses/ DESTINATION share/buggcraft-native/licenses)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/libssh-0.12.2/COPYING")
  install(FILES .deps/libssh-0.12.2/COPYING DESTINATION share/buggcraft-native/licenses RENAME libssh-LGPL-2.1.txt)
endif()
